    class Particle {
        constructor(x, y, color) {
            this.x = x;
            this.y = y;
            const a = Math.random() * Math.PI * 2;
            const s = 2 + Math.random() * 7;
            this.vx = Math.cos(a) * s;
            this.vy = Math.sin(a) * s - 2.5;
            this.life  = 1;
            this.decay = 0.014 + Math.random() * 0.024;
            this.size  = 2 + Math.random() * 5;
            this.color = color;
            this.rot   = Math.random() * Math.PI * 2;
            this.rotV  = (Math.random() - 0.5) * 0.3;
            this.shape = Math.floor(Math.random() * 3); // 0圆 1方 2三角
            this.grav  = 0.07 + Math.random() * 0.06;
        }
        update() {
            this.x  += this.vx;
            this.y  += this.vy;
            this.vy += this.grav;
            this.vx *= 0.98;
            this.life -= this.decay;
            this.rot  += this.rotV;
        }
        draw(c) {
            if (this.life <= 0) return;
            c.save();
            c.translate(this.x, this.y);
            c.rotate(this.rot);
            c.globalAlpha = Math.max(0, this.life);
            c.fillStyle   = this.color;
            const r = Math.max(0.5, this.size * this.life);
            if (this.shape === 0) {
                c.beginPath();
                c.arc(0, 0, r, 0, Math.PI * 2);
                c.fill();
            } else if (this.shape === 1) {
                c.fillRect(-r, -r, r * 2, r * 2);
            } else {
                c.beginPath();
                c.moveTo(0, -r);
                c.lineTo(-r, r);
                c.lineTo(r, r);
                c.closePath();
                c.fill();
            }
            c.restore();
        }
    }

                const a1 = (i * 2 * Math.PI / 5) - Math.PI / 2;

                c.lineTo(Math.cos(a1) * r, Math.sin(a1) * r);
                c.lineTo(Math.cos(a2) * r * 0.4, Math.sin(a2) * r * 0.4);
            }
            c.closePath();
            c.fill();
            c.restore();
        }
    }

            o.connect(g); g.connect(ac.destination);

## 實現說明

這是一個完整的「線上打小人」互動出氣筒，融合了香港驚蟄習俗的傳統元素：

**視覺風格**
- 深紅 + 金色的中國傳統紙扎/廟宇配色，帶有噪點紋理和氛圍光
- SVG 手繪紙扎小人，穿紅色長衫、有盤扣裝飾，細節到位
- 背景有半透明紙人緩緩飄升，底部有裊裊煙霧效果
- 祭壇底座收尾，整體氛圍感強

**核心玩法**
- **4 種武器**：紅拖鞋、報紙卷、雞毛掃、木槌，各有不同音色和打擊感
- **連擊系統**：1.1 秒內連續打擊累計連擊，3 連擊起顯示連擊文字，5 連擊起閃紅光 + 星形粒子，10 連擊起進入「究極暴擊」模式
- **表情變化**：被打時眼睛變 ✕✕、嘴巴變 O 形痛苦表情
- **粒子系統**：金色碎片飛濺，含圓形/方形/三角形/星形四種粒子，連擊越高粒子越密集
- **傳統咒語**：每 3 次或高連擊時隨機顯示正宗打小人咒語

**音效反饋**
- Web Audio API 合成打擊音，每種武器頻率不同
- 連擊時疊加升調提示音
- 無需任何外部音頻文件

**交互細節**
- 支持鼠標、觸屏、鍵盤（Enter/空格）三種操作方式
- 可輸入小人名字顯示在頭頂
- 里程碑 Toast 提示（10/50/100/200/500 下及連擊里程碑）
- 統計卡片在連擊時發光，數值有彈跳動畫
